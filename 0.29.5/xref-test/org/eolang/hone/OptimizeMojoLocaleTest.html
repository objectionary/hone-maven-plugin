5   package org.eolang.hone;
6   
7   import com.yegor256.Jaxec;
8   import java.io.IOException;
9   import java.nio.charset.StandardCharsets;
10  import java.nio.file.Files;
11  import java.nio.file.Paths;
12  import java.util.Arrays;
13  import org.hamcrest.MatcherAssert;
14  import org.hamcrest.Matchers;
15  import org.junit.jupiter.api.Test;
16  
17  /**
18   * Test case for the locale {@code rewrite.sh} sets.
19   *
20   * @since 0.6.1
21   */
22  @SuppressWarnings("JTCOP.RuleEveryTestHasProductionClass")
23  final class OptimizeMojoLocaleTest {
24  
25      @Test
26      void takesTheLocaleTheHostHas() throws IOException {
27          MatcherAssert.assertThat(
28              "the locale the host doesn't have must not be exported",
29              new Jaxec(
30                  "bash", "-c",
31                  String.format(
32                      "set -e%nlocale() { echo 'C.UTF-8'; echo 'POSIX'; }%nLANG=ru_RU.UTF-8%n%s%nprintf '%%s' \"${LANG}\"",
33                      this.paragraph()
34                  )
35              ).exec().stdout().trim(),
36              Matchers.endsWith("C.UTF-8")
37          );
38      }
39  
40      private String paragraph() throws IOException {
41          return Arrays.stream(
42              new String(
43                  Files.readAllBytes(
44                      Paths.get("src/main/resources/org/eolang/hone/scaffolding/rewrite.sh")
45                  ),
46                  StandardCharsets.UTF_8
47              ).split(String.format("%n%n"))
48          ).filter(part -> part.contains("Setting locale to"))
49              .findFirst()
50              .orElseThrow(() -> new IllegalStateException("No locale block found in rewrite.sh"));
51      }
52  }
