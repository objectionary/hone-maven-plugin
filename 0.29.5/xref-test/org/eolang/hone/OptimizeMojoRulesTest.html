104     private static void runOneLargeJnaClass(final Farea fea, final String path,
105         final Path bin, final String image) throws IOException {
106         fea.clean();
107         fea.files()
108             .file(String.format("target/classes/%s", path))
109             .write(Files.readAllBytes(bin));
110         fea.build()
111             .plugins()
112             .appendItself()
113             .execution("default")
114             .phase("process-classes")
115             .goals("build", "optimize")
116             .configuration()
117             .set("alwaysWithDocker", "true")
118             .set("image", image)
119             .set("grepIn", ".*");
120         fea.exec("process-classes");
121         final Path pre = fea.files().file(
122             "target/hone/jeo-disassemble/com/sun/jna/Pointer.xmir"
123         ).path();
124         final Path xmir = fea.files().file(
125             "target/hone/unphi/com/sun/jna/Pointer.xmir"
126         ).path();
127         MatcherAssert.assertThat(
128             "optimized large .xmir must be present",
129             xmir.toFile().exists(),
130             Matchers.is(true)
131         );
132         final Path target = Paths.get(System.getProperty("target.directory"));
133         Files.copy(
134             fea.files().file("target/timings.csv").path(),
135             target.resolve("timings.csv"),
136             StandardCopyOption.REPLACE_EXISTING
137         );
138         final String timing = fea.files().file("target/hone-timings.csv").content();
139         final Matcher mtc = Pattern.compile(
140             String.format("\"optimize\";(?<msec>[0-9]+)%n")
141         ).matcher(timing);
142         MatcherAssert.assertThat(
143             String.format("timing must exist in [%s]", timing),
144             mtc.find(), Matchers.is(true)
145         );
146         final Path phi = fea.files().file(
147             "target/hone/phi/com/sun/jna/Pointer.phi"
148         ).path();
149         final long msec = Long.parseLong(mtc.group("msec"));
150         Files.write(
151             target.resolve("jna-summary.txt"),
152             String.join(
153                 System.lineSeparator(),
154                 String.format("Input: %s", path),
155                 Logger.format(
156                     "Size of .class: %[size]s (%1$s bytes)",
157                     bin.toFile().length()
158                 ),
159                 Logger.format(
160                     "Size of .xmir after disassemble: %[size]s (%1$s bytes, %d lines)",
161                     pre.toFile().length(),
162                     Files.readString(pre, StandardCharsets.UTF_8)
163                         .split(System.lineSeparator()).length
164                 ),
165                 Logger.format(
166                     "Size of .phi: %[size]s (%1$s bytes, %d lines)",
167                     phi.toFile().length(),
168                     Files.readString(phi, StandardCharsets.UTF_8)
169                         .split(System.lineSeparator()).length
170                 ),
171                 Logger.format(
172                     "Size of .xmir after unphi: %[size]s (%1$s bytes, %d lines)",
173                     xmir.toFile().length(),
174                     Files.readString(xmir, StandardCharsets.UTF_8)
175                         .split(System.lineSeparator()).length
176                 ),
177                 Logger.format(
178                     "Optimization time: %[ms]s (%d ms)",
179                     msec, msec
180                 )
181             ).getBytes(StandardCharsets.UTF_8)
182         );
183     }

185     private static void runIncludesAndExcludes(final Farea fea, final String image)
186         throws IOException {
187         fea.clean();
188         fea.files()
189             .file("src/main/java/foo/IncludedClass.java").write(
190                 """
191                 package foo;
192                 class IncludedClass {
193                     int calculate() { return 42; }
194                 }
195                 """.getBytes(StandardCharsets.UTF_8)
196             );
197         fea.files()
198             .file("src/main/java/foo/ExcludedClass.java").write(
199                 """
200                 package foo;
201                 class ExcludedClass {
202                     int calculate() { return 100; }
203                 }
204                 """.getBytes(StandardCharsets.UTF_8)
205             );
206         fea.files()
207             .file("src/main/java/bar/AnotherClass.java").write(
208                 """
209                 package bar;
210                 class AnotherClass {
211                     int calculate() { return 200; }
212                 }
213                 """.getBytes(StandardCharsets.UTF_8)
214             );
215         fea.build()
216             .plugins()
217             .appendItself()
218             .execution("default")
219             .phase("process-classes")
220             .goals("build", "optimize")
221             .configuration()
222             .set("image", image)
223             .set("includes", "/target/classes/foo/Included*")
224             .set("excludes", "/target/classes/foo/Excluded*")
225             .set("grepIn", ".*");
226         fea.exec("process-classes");
227         MatcherAssert.assertThat(
228             "optimized IncludedClass.phi must be present",
229             fea.files().file(
230                 "target/hone/phi-optimized/foo/IncludedClass.phi"
231             ).exists(),
232             Matchers.is(true)
233         );
234         MatcherAssert.assertThat(
235             "ExcludedClass.phi must not be optimized",
236             fea.files().file(
237                 "target/hone/phi-optimized/foo/ExcludedClass.phi"
238             ).exists(),
239             Matchers.is(false)
240         );
241         MatcherAssert.assertThat(
242             "AnotherClass.phi must not be optimized (not included)",
243             fea.files().file(
244                 "target/hone/phi-optimized/bar/AnotherClass.phi"
245             ).exists(),
246             Matchers.is(false)
247         );
248     }

274                 """.getBytes(StandardCharsets.UTF_8)

292                     }
293                 """.getBytes(StandardCharsets.UTF_8)
294             );

297                 """
298                 import org.junit.jupiter.api.Assertions;
299                 import org.junit.jupiter.api.Test;
300                 class FooTest {
301                     @Test
302                     void worksAfterOptimizationWithExtraRule() {
303                         Assertions.assertEquals(
304                             3531468,
305                             new Foo().bar()
306                         );
307                     }
308                 }
309                 """.getBytes(StandardCharsets.UTF_8)
310             );

312             .append("org.junit.jupiter", "junit-jupiter-engine", "5.10.2");
313         fea.dependencies()
314             .append("org.junit.jupiter", "junit-jupiter-params", "5.10.2");
315         fea.build()
316             .plugins()
317             .appendItself()
318             .execution("default")
319             .phase("process-classes")
320             .goals("build", "optimize")
321             .configuration()
322             .set("rules", "none")
323             .set("threads", "1")
324             .set("smallSteps", "true")
325             .set("maxDepth", "10").set(
326                 "extra",
327                 new String[] {
328                     "src/rules/first.yaml",
329                     "src/rules/second.yaml",
330                     "src/rules/a-few",
331                 }
332             )
333             .set("image", image)
334             .set("grepIn", ".*");
335         fea.exec("test");
336         MatcherAssert.assertThat(
337             "the build must be successful",
338             fea.log(),
339             RequisiteMatcher.SUCCESS
340         );
341     }
