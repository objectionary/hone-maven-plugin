13  import java.nio.file.Path;
14  import java.nio.file.Paths;
15  import java.util.regex.Matcher;
16  import java.util.regex.Pattern;
17  import org.hamcrest.MatcherAssert;
18  import org.hamcrest.Matchers;
19  import org.junit.jupiter.api.Test;
20  import org.junit.jupiter.api.extension.ExtendWith;
21  
22  /**
23   * Test case for the count {@link OptimizeMojo} prints through {@code entry.sh}.
24   *
25   * @since 0.6.1
26   */
27  @ExtendWith(MktmpResolver.class)
28  @SuppressWarnings("JTCOP.RuleEveryTestHasProductionClass")
29  final class OptimizeMojoSavedCountTest {
30  
31      @Test
32      void countsOnlyFilesNotDirectories(@Mktmp final Path temp) throws IOException {
33          final Path saved = temp.resolve("classes-before-hone");
34          final Path pkg = saved.resolve("org/eolang/deep");
35          Files.createDirectories(pkg);
36          for (final String name : new String[] {"One.class", "Two.class", "Three.class"}) {
37              Files.write(pkg.resolve(name), "x".getBytes(StandardCharsets.UTF_8));
38          }
39          final Matcher matcher = Pattern.compile(
40              "echo \"The binaries before hone are saved.*"
41          ).matcher(
42              new String(
43                  Files.readAllBytes(
44                      Paths.get("src/main/resources/org/eolang/hone/scaffolding/entry.sh")
45                  ),
46                  StandardCharsets.UTF_8
47              )
48          );
49          if (!matcher.find()) {
50              throw new IllegalStateException("No saving message found in entry.sh");
51          }
52          MatcherAssert.assertThat(
53              "three class files in three nested directories must be counted as three",
54              new Jaxec("bash", "-c", matcher.group())
55                  .withEnv("TARGET", temp.toString())
56                  .exec()
57                  .stdout(),
58              Matchers.containsString("(3 files)")
59          );
60      }
61  }
