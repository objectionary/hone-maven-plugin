5   package org.eolang.hone;
6   
7   import com.yegor256.Jaxec;
8   import java.nio.charset.StandardCharsets;
9   import java.nio.file.Files;
10  import java.nio.file.Path;
11  import java.nio.file.Paths;
12  import org.hamcrest.MatcherAssert;
13  import org.hamcrest.Matchers;
14  import org.junit.jupiter.api.Test;
15  
16  /**
17   * Test case for the per-file timeout of {@code rewrite.sh}.
18   *
19   * @since 0.6.1
20   */
21  @SuppressWarnings("JTCOP.RuleEveryTestHasProductionClass")
22  final class OptimizeMojoTimeoutTest {
23  
24      @Test
25      void sweepsTempFilesOfKilledWorker() throws Exception {
26          try (Mktemp temp = new Mktemp()) {
27              final Path home = temp.path();
28              final Path phino = home.resolve("phino");
29              Files.write(
30                  phino,
31                  String.format("#!/usr/bin/env bash%necho partial%nsleep 30%n")
32                      .getBytes(StandardCharsets.UTF_8)
33              );
34              if (!phino.toFile().setExecutable(true)) {
35                  throw new IllegalStateException("Can't make the fake phino executable");
36              }
37              Files.write(home.resolve("in.xmir"), "<xmir/>".getBytes(StandardCharsets.UTF_8));
38              MatcherAssert.assertThat(
39                  "a worker killed by the timeout must not leave its temp files behind",
40                  new Jaxec(
41                      "bash", "-c",
42                      String.format(
43                          String.join(
44                              String.format("%n"),
45                              "set -e",
46                              "export PATH=%1$s:${PATH} TARGET=%1$s HONE_RULES=%1$s/simple.phr",
47                              "export HONE_MAX_CYCLES=1 HONE_MAX_DEPTH=1 HONE_SMALL_STEPS=false",
48                              "export HONE_STATISTICS=false HONE_VERBOSE=false HONE_DEBUG=false",
49                              "export HONE_TIMEOUT=1 HONE_KILL_GRACE=1",
50                              "cd %1$s",
51                              "bash %2$s rewrite_with_timeout 1/1 a/a.phi b/a.phi in.xmir c/a.xmir > /dev/null",
52                              "find %1$s -name '*.tmp.*'"
53                          ),
54                          home,
55                          Paths.get(
56                              "src/main/resources/org/eolang/hone/scaffolding/rewrite.sh"
57                          ).toAbsolutePath()
58                      )
59                  ).exec().stdout(),
60                  Matchers.emptyString()
61              );
62          }
63      }
64  }
