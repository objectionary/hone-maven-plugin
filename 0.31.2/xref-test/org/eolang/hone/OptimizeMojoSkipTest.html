9   import com.yegor256.MktmpResolver;
10  import java.io.IOException;
11  import java.nio.charset.StandardCharsets;
12  import java.nio.file.Files;
13  import java.nio.file.Path;
14  import java.nio.file.Paths;
15  import java.util.regex.Matcher;
16  import java.util.regex.Pattern;
17  import org.hamcrest.MatcherAssert;
18  import org.hamcrest.Matchers;
19  import org.junit.jupiter.api.Test;
20  import org.junit.jupiter.api.extension.ExtendWith;
21  
22  /**
23   * Test case for the skip check of {@code rewrite.sh}, as {@code entry.sh} runs it.
24   *
25   * @since 0.6.1
26   */
27  @ExtendWith(MktmpResolver.class)
28  @SuppressWarnings("JTCOP.RuleEveryTestHasProductionClass")
29  final class OptimizeMojoSkipTest {
30  
31      @Test
32      void skipsUnchangedClassOnSecondRun(@Mktmp final Path temp) throws IOException {
33          MatcherAssert.assertThat(
34              "a class that did not change since the previous run must be skipped",
35              this.twice(temp, "printf '<xmir/>' > hone/jeo-disassemble/A.xmir"),
36              Matchers.containsString("skipping transformation for 1/1")
37          );
38      }
39  
40      @Test
41      void dropsOutputOfClassThatIsGone(@Mktmp final Path temp) throws IOException {
42          this.twice(temp, "printf '<xmir/>' > hone/jeo-disassemble/B.xmir");
43          MatcherAssert.assertThat(
44              "the output of a class that is no longer disassembled must not survive",
45              temp.resolve("hone/unphi/A.xmir").toFile().exists(),
46              Matchers.is(false)
47          );
48      }
49  
50      private String twice(final Path home, final String between) throws IOException {
51          final Path phino = home.resolve("phino");
52          Files.write(
53              phino,
54              String.format(
55                  "#!/usr/bin/env bash%n[ \"${1}\" == '--version' ] && echo 0.0.1 && exit%ncat \"${@: -1}\"%n"
56              ).getBytes(StandardCharsets.UTF_8)
57          );
58          if (!phino.toFile().setExecutable(true)) {
59              throw new IllegalStateException("Can't make the fake phino executable");
60          }
61          Files.write(home.resolve("simple.phr"), "rule".getBytes(StandardCharsets.UTF_8));
62          Files.createDirectories(home.resolve("hone/jeo-disassemble"));
63          Files.write(
64              home.resolve("hone/jeo-disassemble/A.xmir"),
65              "<xmir/>".getBytes(StandardCharsets.UTF_8)
66          );
67          final Matcher matcher = Pattern.compile("rm -rf \"\\$\\{TARGET\\}/hone/[^\\n]*\\n").matcher(
68              new String(
69                  Files.readAllBytes(
70                      Paths.get("src/main/resources/org/eolang/hone/scaffolding/entry.sh")
71                  ),
72                  StandardCharsets.UTF_8
73              )
74          );
75          if (!matcher.find()) {
76              throw new IllegalStateException("No clearing of the pipeline directories in entry.sh");
77          }
78          return new Jaxec(
79              "bash", "-c",
80              String.format(
81                  String.join(
82                      String.format("%n"),
83                      "set -e",
84                      "cd %1$s",
85                      "export PATH=%1$s:${PATH} TARGET=%1$s HONE_RULES=%1$s/simple.phr",
86                      "export HONE_XMIR_IN=hone/jeo-disassemble HONE_XMIR_OUT=hone/unphi",
87                      "export HONE_FROM=hone/phi HONE_TO=hone/phi-optimized HONE_THREADS=1",
88                      "export HONE_MAX_CYCLES=1 HONE_MAX_DEPTH=1 HONE_SMALL_STEPS=false",
89                      "export HONE_STATISTICS=false HONE_VERBOSE=false HONE_DEBUG=false HONE_TIMEOUT=5",
90                      "bash %2$s > /dev/null",
91                      "%3$smkdir -p hone/jeo-disassemble",
92                      "%4$s",
93                      "bash %2$s"
94                  ),
95                  home,
96                  Paths.get(
97                      "src/main/resources/org/eolang/hone/scaffolding/rewrite.sh"
98                  ).toAbsolutePath(),
99                  matcher.group(),
100                 between
101             )
102         ).exec().stdout();
103     }
