208     @Test
209     void reachesEveryProductionOfTheGrammar() {
210         final StringBuilder walked = new StringBuilder();
211         for (int seed = 0; seed < 2000; ++seed) {
212             walked.append(new RandomPipeline(seed).java("reached", "X"));
213         }
214         final List<String> missed = new ArrayList<>(0);
215         final Grammar grammar = new Grammar();
216         for (final String production : grammar.fragments()) {
217             if (!walked.toString().contains(Grammar.fragment(production))) {
218                 missed.add(production);
219             }
220         }
221         MatcherAssert.assertThat(
222             "every production must be reachable, or the grammar promises more than it walks",
223             missed,
224             Matchers.empty()
225         );
226     }

228     @Test
229     void generatesPipelinesThatCompile(@Mktmp final Path dir) throws IOException {
230         final List<String> files = new ArrayList<>(0);
231         for (int seed = 0; seed < 500; ++seed) {
232             final String name = String.format("P%04d", seed);
233             final Path java = dir.resolve(String.format("%s.java", name));
234             Files.write(
235                 java,
236                 new RandomPipeline(seed).java("compiled", name).getBytes(StandardCharsets.UTF_8)
237             );
238             files.add(java.toString());
239         }
240         final ByteArrayOutputStream errors = new ByteArrayOutputStream();
241         final List<String> args = new ArrayList<>(0);
242         args.add("-d");
243         args.add(dir.toString());
244         args.addAll(files);
245         ToolProvider.getSystemJavaCompiler().run(
246             null, null, errors, args.toArray(new String[0])
247         );
248         MatcherAssert.assertThat(
249             "every generated class must compile, or the grammar is not typed",
250             new String(errors.toByteArray(), StandardCharsets.UTF_8),
251             Matchers.emptyString()
252         );
253     }

263         final int pipelines = Integer.getInteger("hone.random.pipelines", 120);

265             f -> RandomPipelineOptimizationTest.runRandomPipelines(f, home, pipelines, image)

269     private static void runRandomPipelines(final Farea fea, final Path home,
270         final int pipelines, final String image) throws IOException {
271         fea.clean();
272         fea.properties().set("maven.compiler.release", "17");
273         for (int seed = 0; seed < pipelines; ++seed) {
274             final String name = String.format("P%04d", seed);
275             fea.files()
276                 .file(String.format("src/main/java/random/%s.java", name)).write(
277                     new RandomPipeline(seed)
278                         .java("random", name)
279                         .getBytes(StandardCharsets.UTF_8)
280                 );
281         }
282         fea.build()
283             .plugins()
284             .appendItself()
285             .execution("default")
286             .phase("process-classes")
287             .goals("build", "optimize")
288             .configuration()
289             .set("rules", "streams/*")
290             .set("grepIn", ".*")
291             .set("image", image);
292         fea.exec("process-classes");
293         MatcherAssert.assertThat(
294             "the build of the random pipelines must be successful",
295             fea.log(),
296             RequisiteMatcher.SUCCESS
297         );
298         int rewritten = 0;
299         for (int seed = 0; seed < pipelines; ++seed) {
300             final String klass = String.format("random/P%04d.class", seed);
301             if (!Arrays.equals(
302                 Files.readAllBytes(home.resolve("target/classes").resolve(klass)),
303                 Files.readAllBytes(
304                     home.resolve("target/classes-before-hone").resolve(klass)
305                 )
306             )) {
307                 rewritten += 1;
308             }
309         }
310         MatcherAssert.assertThat(
311             String.format(
312                 "at least one of the %d pipelines must be rewritten, or the experiment proves nothing",
313                 pipelines
314             ),
315             rewritten,
316             Matchers.greaterThan(0)
317         );
318         final List<String> broken = new ArrayList<>(0);
319         for (int seed = 0; seed < pipelines; ++seed) {
320             final String name = String.format("P%04d", seed);
321             final Result before = RandomPipelineOptimizationTest.runs(
322                 home, "target/classes-before-hone", name
323             );
324             final Result after = RandomPipelineOptimizationTest.runs(home, "target/classes", name);
325             final String complaint = RandomPipelineOptimizationTest.differs(
326                 before, after, new RandomPipeline(seed).java("random", name), name
327             );
328             if (!complaint.isEmpty()) {
329                 broken.add(complaint);
330             }
331         }
332         MatcherAssert.assertThat(
333             String.format(
334                 "optimization must not change what a pipeline prints, %d of %d did",
335                 broken.size(), pipelines
336             ),
337             broken,
338             Matchers.empty()
339         );
340     }

342     private static String differs(final Result before, final Result after,
343         final String java, final String name) {
344         final String complaint;
345         if (before.code() == 0 && after.code() == 0
346             && before.stdout().equals(after.stdout())) {
347             complaint = "";
348         } else if (before.code() == 0 && after.code() == 0) {
349             complaint = String.format(
350                 "%s printed '%s' before optimization and '%s' after it:%n%s",
351                 name, before.stdout().trim(), after.stdout().trim(), java
352             );
353         } else if (before.code() == 0) {
354             complaint = String.format(
355                 "%s cannot run after optimization:%n%s%n%s", name, java, after.stderr()
356             );
357         } else {
358             complaint = String.format(
359                 "%s cannot run before optimization, so the grammar is broken:%n%s%n%s",
360                 name, java, before.stderr()
361             );
362         }
363         return complaint;
364     }

366     private static Result runs(final Path home, final String dir, final String name)
367         throws IOException {
368         return new Jaxec(
369             new Jhome().java().toString(),
370             "-cp",
371             home.resolve(dir).toString(),
372             String.format("random.%s", name)
373         ).withCheck(false).execUnsafe();
374     }
