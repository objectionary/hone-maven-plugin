5   package org.eolang.hone;
6   
7   import com.yegor256.Jaxec;
8   import java.io.IOException;
9   import java.nio.charset.StandardCharsets;
10  import java.nio.file.Files;
11  import java.nio.file.Paths;
12  import java.util.regex.Matcher;
13  import java.util.regex.Pattern;
14  import org.hamcrest.MatcherAssert;
15  import org.hamcrest.Matchers;
16  import org.junit.jupiter.api.Test;
17  
18  /**
19   * Test case for the rule name {@link OptimizeMojo} prints in small steps mode.
20   *
21   * @since 0.6.1
22   */
23  @SuppressWarnings("JTCOP.RuleEveryTestHasProductionClass")
24  final class OptimizeMojoRuleNameTest {
25  
26      @Test
27      void stripsTheExtensionOfAPhinoRule() throws IOException {
28          final Matcher matcher = Pattern.compile(
29              "m=\\$\\(basename.*?\\n(\\s*m=\"\\$\\{m%\\.\\*\\}\"\\n)?",
30              Pattern.DOTALL
31          ).matcher(
32              new String(
33                  Files.readAllBytes(
34                      Paths.get("src/main/resources/org/eolang/hone/scaffolding/rewrite.sh")
35                  ),
36                  StandardCharsets.UTF_8
37              )
38          );
39          if (!matcher.find()) {
40              throw new IllegalStateException("No rule name stripping found in rewrite.sh");
41          }
42          MatcherAssert.assertThat(
43              "the extension of a phino rule must be stripped from the name that is printed",
44              new Jaxec(
45                  "bash", "-c",
46                  String.format(
47                      "rule=/x/101-remove-self-reference-labels.phr%n%sprintf '%%s' \"${m}\"",
48                      matcher.group()
49                  )
50              ).exec().stdout().trim(),
51              Matchers.equalTo("101-remove-self-reference-labels")
52          );
53      }
54  }
