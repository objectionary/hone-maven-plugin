7   import com.yegor256.Jaxec;

9   import com.yegor256.MktmpResolver;
10  import java.io.IOException;
11  import java.nio.charset.StandardCharsets;
12  import java.nio.file.Files;
13  import java.nio.file.Path;
14  import java.nio.file.Paths;
15  import java.util.regex.Matcher;
16  import java.util.regex.Pattern;
17  import org.hamcrest.MatcherAssert;
18  import org.hamcrest.Matchers;
19  import org.junit.jupiter.api.Test;
20  import org.junit.jupiter.api.extension.ExtendWith;
21  
22  /**
23   * Test case for the directories {@link OptimizeMojo} clears through {@code entry.sh}.
24   *
25   * @since 0.6.1
26   */
27  @ExtendWith(MktmpResolver.class)
28  @SuppressWarnings("JTCOP.RuleEveryTestHasProductionClass")
29  final class OptimizeMojoStaleXmirTest {
30  
31      @Test
32      void clearsTheXmirOfThePreviousRun(@Mktmp final Path temp) throws IOException {
33          final Path stale = temp.resolve("hone/jeo-disassemble/Gone.xmir");
34          Files.createDirectories(stale.getParent());
35          Files.write(stale, "x".getBytes(StandardCharsets.UTF_8));
36          final Matcher matcher = Pattern.compile("rm -rf \"\\$\\{TARGET\\}/hone/[^\\n]*\\n").matcher(
37              new String(
38                  Files.readAllBytes(
39                      Paths.get("src/main/resources/org/eolang/hone/scaffolding/entry.sh")
40                  ),
41                  StandardCharsets.UTF_8
42              )
43          );
44          if (!matcher.find()) {
45              throw new IllegalStateException("No clearing of the pipeline directories in entry.sh");
46          }
47          new Jaxec("bash", "-c", matcher.group()).withEnv("TARGET", temp.toString()).exec();
48          MatcherAssert.assertThat(
49              "XMIR of a class that is no longer disassembled cannot survive into the next run",
50              stale.toFile().exists(),
51              Matchers.is(false)
52          );
53      }
54  }
