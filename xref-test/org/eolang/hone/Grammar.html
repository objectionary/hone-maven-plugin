14  /**
15   * The grammar of the Stream API that {@link RandomPipeline} walks.
16   *
17   * <p>One production per line, as {@code domain|role|fragment}. A {@code turn}
18   * fragment carries the domain it lands in as a fourth field, and an {@code end}
19   * fragment is prefixed with {@code *} when the API pins how much of the stream
20   * it walks. Nothing here is {@code findAny}: the value a pipeline prints has to
21   * be the same on every run, or it cannot be an oracle. Nothing here says
22   * {@code parallel()} either, because that is a property of a pipeline and not of
23   * a step in it — the walk appends it once, at the end, where it means the same
24   * thing as it would anywhere else.</p>
25   *
26   * <p>Four roles beyond the plain ones: a {@code peek} counts the traversal, a
27   * {@code self} reads {@code this} and is drawn only in an instance frame, a
28   * {@code boom} throws and is drawn only in a frame that catches, and a
29   * {@code void} terminal returns nothing and leaves the peek line as the only
30   * oracle.</p>
31   *
32   * <p>The productions reach the three primitive domains, all three of their boxed
33   * counterparts — {@code Stream<Long>}, {@code Stream<Integer>} and
34   * {@code Stream<Double>} — and two reference domains, one of strings and one of
35   * a small value class of its own. They reach the corners of the terminal API as
36   * well: collectors assembled by hand out of {@code Collector.of},
37   * {@code Collectors.teeing}, {@code Collectors.flatMapping}, and
38   * {@code Optional} chains longer than a single {@code orElse}.</p>
39   *
40   * <p>A production wider than one line is joined from pieces, which is what the
41   * few {@code String.join} entries are. The terminals that tee, and the ones that
42   * assemble a {@code Collector} by hand, spell their type arguments out: a
43   * pipeline is printed through {@code String.valueOf}, and its overloads bound
44   * the result of such a terminal by {@code char[]} as well as by {@code Object},
45   * which no inferred type can satisfy.</p>
46   *
47   * @since 0.30.0
48   */
49  final class Grammar {

441     /**
442      * One domain to start a pipeline in.
443      * @param rnd The source of randomness
444      * @return The domain, as the grammar names it
445      */
446     static String domain(final Random rnd) {
447         return Grammar.DOMAINS.get(rnd.nextInt(Grammar.DOMAINS.size()));
448     }

450     /**
451      * Every fragment the grammar can reach, whatever domain it belongs to.
452      * @return The fragments, as they are written in the grammar
453      */
454     List<String> fragments() {
455         final List<String> all = new ArrayList<>(0);
456         for (final List<String> allowed : this.productions.values()) {
457             all.addAll(allowed);
458         }
459         return all;
460     }

462     /**
463      * One fragment of the given domain and role.
464      * @param domain The element domain the pipeline is in
465      * @param role What the fragment must be: a source, a stage, a turn, a peek
466      * @param rnd The source of randomness
467      * @return The fragment, verbatim from the grammar
468      */
469     String pick(final String domain, final String role, final Random rnd) {
470         final List<String> allowed = this.productions.get(String.join("|", domain, role));
471         return allowed.get(rnd.nextInt(allowed.size()));
472     }
473 
474     /**
475      * The terminal to end the pipeline with.
476      *
477      * <p>A pipeline whose traversal is counted — by a {@code peek}, or by a
478      * {@code boom} that names the element it threw on — may not end in a
479      * terminal the API lets skip the traversal, which is {@code count()} and
480      * nothing else. Short-circuiting terminals stay allowed on purpose: how far
481      * a sequential stream walks before it stops is part of what the rewrite has
482      * to preserve. A terminal that returns nothing walks everything by
483      * contract, so it is offered whatever else the walk did.</p>
484      *
485      * @param domain The element domain the pipeline ended in
486      * @param observed TRUE if something counts the traversal
487      * @param rnd The source of randomness
488      * @return The terminal line, as {@code domain|role|fragment}
489      */
490     String terminal(final String domain, final boolean observed, final Random rnd) {
491         final List<String> ends = new ArrayList<>(0);
492         for (final String end : this.productions.get(String.join("|", domain, "end"))) {
493             if (!observed || end.charAt(0) == '*') {
494                 ends.add(String.join("|", domain, "end", Grammar.fragment(end)));
495             }
496         }
497         for (final String end : this.productions.get(String.join("|", domain, "void"))) {
498             ends.add(String.join("|", domain, "void", end));
499         }
500         return ends.get(rnd.nextInt(ends.size()));
501     }

503     /**
504      * The Java code of one line of a walk.
505      * @param line The line, as {@code domain|role|fragment}
506      * @return What the pipeline appends for it
507      */
508     static String code(final String line) {
509         return Grammar.fragment(line.split("\\|", 3)[2]);
510     }
511 
512     /**
513      * The Java code of a fragment, without the marks the grammar wraps it in.
514      * @param production The fragment, as it is written in the grammar
515      * @return What the walk appends to a pipeline when it picks this fragment
516      */
517     static String fragment(final String production) {
518         String code = production;
519         final int lands = code.lastIndexOf('|');
520         if (lands >= 0 && Grammar.DOMAINS.contains(code.substring(lands + 1))) {
521             code = code.substring(0, lands);
522         }
523         if (code.charAt(0) == '*') {
524             code = code.substring(1);
525         }
526         return code;
527     }
528 
529     /**
530      * The productions, indexed by domain and role.
531      * @return Map from {@code domain|role} to the fragments it allows
532      */
533     private static Map<String, List<String>> indexed() {
534         final Map<String, List<String>> table = new HashMap<>();
535         for (final String line : Grammar.LINES) {
536             final String[] parts = line.split("\\|", 3);
537             table.computeIfAbsent(
538                 String.join("|", parts[0], parts[1]), key -> new ArrayList<>(0)
539             ).add(parts[2]);
540         }
541         return table;
542     }
