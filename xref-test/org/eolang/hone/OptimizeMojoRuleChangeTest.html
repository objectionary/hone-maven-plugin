5   package org.eolang.hone;
6   
7   import com.yegor256.Jaxec;
8   import java.nio.charset.StandardCharsets;
9   import java.nio.file.Files;
10  import java.nio.file.Path;
11  import java.nio.file.Paths;
12  import org.hamcrest.MatcherAssert;
13  import org.hamcrest.Matchers;
14  import org.junit.jupiter.api.Test;
15  
16  /**
17   * Test case for the skip check of {@code rewrite.sh}.
18   *
19   * @since 0.6.1
20   */
21  @SuppressWarnings("JTCOP.RuleEveryTestHasProductionClass")
22  final class OptimizeMojoRuleChangeTest {
23  
24      @Test
25      void rewritesAgainAfterTheRuleIsEdited() throws Exception {
26          try (Mktemp temp = new Mktemp()) {
27              final Path home = temp.path();
28              final Path phino = home.resolve("phino");
29              Files.write(
30                  phino,
31                  String.format("#!/usr/bin/env bash%ncat \"${@: -1}\"%n")
32                      .getBytes(StandardCharsets.UTF_8)
33              );
34              if (!phino.toFile().setExecutable(true)) {
35                  throw new IllegalStateException("Can't make the fake phino executable");
36              }
37              Files.write(home.resolve("simple.phr"), "rule".getBytes(StandardCharsets.UTF_8));
38              Files.write(home.resolve("in.xmir"), "<xmir/>".getBytes(StandardCharsets.UTF_8));
39              MatcherAssert.assertThat(
40                  "an edited rule must not be skipped",
41                  new Jaxec(
42                      "bash", "-c",
43                      String.format(
44                          String.join(
45                              String.format("%n"),
46                              "set -e",
47                              "export PATH=%1$s:${PATH} TARGET=%1$s HONE_RULES=%1$s/simple.phr",
48                              "export HONE_MAX_CYCLES=1 HONE_MAX_DEPTH=1 HONE_SMALL_STEPS=false",
49                              "export HONE_STATISTICS=false HONE_VERBOSE=false HONE_DEBUG=false",
50                              "run() { bash %2$s rewrite 1/1 %1$s/a.phi %1$s/b.phi %1$s/in.xmir %1$s/out.xmir; }",
51                              "run > /dev/null",
52                              "touch -t 200001010000 %1$s/simple.phr",
53                              "run"
54                          ),
55                          home,
56                          Paths.get(
57                              "src/main/resources/org/eolang/hone/scaffolding/rewrite.sh"
58                          ).toAbsolutePath()
59                      )
60                  ).exec().stdout(),
61                  Matchers.not(Matchers.containsString("skipping"))
62              );
63          }
64      }
65  }
